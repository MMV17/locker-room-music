# Test-only dependencies. NOT installed on the Pi — provision.sh deliberately
# installs requirements.txt and nothing else, because the box should carry the
# code it runs and no more.
#
# pytest-asyncio is the load-bearing line here: without it every async test
# FAILS rather than skipping, with the only clue a PytestUnknownMarkWarning
# buried in the warning summary. That reads exactly like fifty broken tests.
#
#   python3 -m venv .venv && .venv/bin/pip install -r pi/requirements.txt -r pi/requirements-dev.txt
#   cd pi && ../.venv/bin/python -m pytest tests -q
#
# pi/pyproject.toml puts pi/ on the import path and configures ruff
# (`ruff check` from pi/). The updater tests need a real `git` on PATH.
pytest>=8
pytest-asyncio>=0.23
